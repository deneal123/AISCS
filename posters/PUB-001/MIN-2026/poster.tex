% Each row contains 98 unit cells. Positions show counts, not study identities.
\newcommand{\ValidationRow}[4]{%
  \pgfmathsetmacro{\PercentValue}{100*(#3)/\NModels}
  \pgfmathtruncatemacro{\GridColumns}{\NModels/2}
  \pgfmathsetmacro{\CellPitch}{\ChartWidth/\GridColumns}
  \TextAt{\ChartX}{#1}{165}{\BodyFont}{#2}
  \node[anchor=north east,inner sep=0pt,outer sep=0pt,text=MINink,
    font=\fontsize{24}{29}\selectfont] at (\ChartRight,#1)
    {\textbf{#3/\NModels}\;·\;\pgfmathprintnumber[
      fixed,precision=1,fixed zerofill,use comma,assume math mode=true]
      {\PercentValue}\,\%};
  \foreach \study in {1,...,\NModels} {
    \pgfmathtruncatemacro{\CellColumn}{mod(\study-1,\GridColumns)}
    \pgfmathtruncatemacro{\CellRow}{floor((\study-1)/\GridColumns)}
    \ifnum\study>#3\relax
      \def\CellColour{MINline}
    \else
      \def\CellColour{#4}
    \fi
    \fill[\CellColour]
      ({\ChartX+\CellColumn*\CellPitch},{#1+12+\CellRow*4.1}) rectangle
      ({\ChartX+(\CellColumn+1)*\CellPitch-0.7},{#1+15.4+\CellRow*4.1});
  }}
